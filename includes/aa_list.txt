<?php
declare(strict_types=1);

const AA_LIST_CACHE_TTL = 3600;

function aaListCacheFile(
    string $domain
): string {

    $dir =
        __DIR__ .
        '/../cache/aa';

    if (!is_dir($dir)) {
        mkdir(
            $dir,
            0775,
            true
        );
    }

    $safeDomain =
        preg_replace(
            '/[^a-zA-Z0-9._-]/',
            '_',
            $domain
        );

    return
        $dir .
        '/' .
        ($safeDomain ?: 'default') .
        '-list.json';
}

function loadAaListCache(
    string $domain
): ?array {

    $file =
        aaListCacheFile($domain);

    if (
        !is_file($file)
    ) {
        return null;
    }

    if (
        (time() - filemtime($file))
        > AA_LIST_CACHE_TTL
    ) {
        return null;
    }

    $json =
        file_get_contents($file);

    if (
        $json === false ||
        trim($json) === ''
    ) {
        return null;
    }

    $data =
        json_decode(
            $json,
            true
        );

    return
        is_array($data)
            ? $data
            : null;
}

function saveAaListCache(
    string $domain,
    array $data
): void {

    $file =
        aaListCacheFile($domain);

    $tmp =
        $file . '.tmp';

    $json =
        json_encode(
            $data,
            JSON_PRETTY_PRINT |
            JSON_UNESCAPED_SLASHES
        );

    if ($json === false) {
        return;
    }

    if (
        file_put_contents(
            $tmp,
            $json,
            LOCK_EX
        ) === false
    ) {
        return;
    }

    rename(
        $tmp,
        $file
    );
}

function getAaList(
    string $domain,
    PbxClient $client
): array {

    appLog(
        'AA LIST: START ' . $domain
    );

    $cached =
        loadAaListCache(
            $domain
        );

    if (
        $cached !== null
    ) {

        appLog(
            'AA LIST: CACHE HIT ' .
            $domain .
            ' COUNT=' .
            count($cached)
        );

        return $cached;
    }

    appLog(
        'AA LIST: CACHE MISS ' .
        $domain
    );


    try {

        $rows =
            $client->autoAttendantList(
                $domain
            );

    } catch (Throwable $e) {

        appLog(
            'AA LIST: API EXCEPTION ' .
            $domain .
            ': ' .
            $e->getMessage()
        );

        return [];
    }


    appLog(
        'AA LIST: API RESULT ' .
        $domain .
        ' = ' .
        json_encode(
            $rows,
            JSON_UNESCAPED_SLASHES
        )
    );


    $result = [];

    foreach ($rows as $row) {

        if (!is_array($row)) {
            continue;
        }

        $extension =
            trim(
                (string)(
                    $row['user'] ?? ''
                )
            );

        $startingPrompt =
            trim(
                (string)(
                    $row['starting-prompt'] ?? ''
                )
            );

        if (
            $extension === '' ||
            $startingPrompt === ''
        ) {
            continue;
        }

        $result[$extension] = [
            'extension' =>
                $extension,

            'name' =>
                trim(
                    (string)(
                        $row['attendant-name'] ?? ''
                    )
                ),

            'startingPrompt' =>
                $startingPrompt,

            'readPath' =>
                trim(
                    (string)(
                        $row['read-path'] ?? ''
                    )
                ),

            'timeframes' =>
                is_array(
                    $row['timeframes'] ?? null
                )
                    ? $row['timeframes']
                    : []
        ];
    }


    appLog(
        'AA LIST: NORMALIZED COUNT=' .
        count($result)
    );


    saveAaListCache(
        $domain,
        $result
    );


    appLog(
        'AA LIST: CACHE SAVE COMPLETE ' .
        $domain
    );


    return $result;
}
